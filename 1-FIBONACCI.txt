def fibItr(n):
    prevNum = 0
    currentNum = 1

    if n == 0:
        return 0

    for i in range(1, n):
        prevPrevNum = prevNum
        prevNum = currentNum
        currentNum = prevPrevNum + prevNum

    return currentNum


def fibRecur(n):
    if n == 0:
        return 0
    elif n == 1:
        return 1
    else:
        return fibRecur(n - 1) + fibRecur(n - 2)


if __name__ == "__main__":
    num = int(input("Enter a no. : "))

    print(f"Using Recursion value of fib({num}) is {fibRecur(num)}")
    print(f"Using Iteration value of fib({num}) is {fibItr(num)}")

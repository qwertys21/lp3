import pandas as pd
import matplotlib.pyplot as plt

from sklearn.preprocessing import StandardScaler
from sklearn.cluster import KMeans

# Step 1: Load the dataset
df = pd.read_csv("sales_data_sample.csv", encoding="latin1")

print("Dataset:")
print(df.head())

# Step 2: Select numerical features
X = df[["QUANTITYORDERED", "PRICEEACH", "SALES", "MSRP"]]

# Step 3: Scale the data
scaler = StandardScaler()
X_scaled = scaler.fit_transform(X)

# Step 4: Find the number of clusters using Elbow Method
wcss = []

for i in range(1, 11):
    kmeans = KMeans(n_clusters=i, random_state=42, n_init=10)
    kmeans.fit(X_scaled)
    wcss.append(kmeans.inertia_)

# Step 5: Plot the Elbow graph
plt.plot(range(1, 11), wcss, marker="o")
plt.title("Elbow Method")
plt.xlabel("Number of Clusters")
plt.ylabel("WCSS")
plt.show()

# Step 6: Apply K-Means with 3 clusters
kmeans = KMeans(n_clusters=3, random_state=42, n_init=10)

df["Cluster"] = kmeans.fit_predict(X_scaled)

# Step 7: Display the results
print("\nCluster Results:")
print(df[["QUANTITYORDERED", "PRICEEACH", "SALES", "MSRP", "Cluster"]].head())

print("\nNumber of records in each cluster:")
print(df["Cluster"].value_counts())

print("\nCluster Centers:")
print(scaler.inverse_transform(kmeans.cluster_centers_))
